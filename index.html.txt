    /* 🎯 스마트 붙여넣기 처리: 링크(<a> 태그) 및 URL 자동 하이퍼링크 보존 */
    document.addEventListener('paste', function(e) {
      if (e.target.isContentEditable) {
        e.preventDefault();
        let clipboardData = e.clipboardData || window.clipboardData;
        let htmlData = clipboardData.getData('text/html');
        let textData = clipboardData.getData('text/plain');

        let container = document.createElement('div');
        if (htmlData) {
          container.innerHTML = htmlData;
          cleanHTMLForPaste(container);
        } else if (textData) {
          container.innerHTML = autoLinkText(textData);
        }

        let sel = window.getSelection();
        if (sel.rangeCount) {
          let range = sel.getRangeAt(0);
          range.deleteContents();
          let fragment = document.createDocumentFragment();
          let child;
          while ((child = container.firstChild)) {
            fragment.appendChild(child);
          }
          range.insertNode(fragment);
        }
      }
    });

    /* 텍스트 내 URL 자동 감지 및 <a> 변환 */
    function autoLinkText(text) {
      const urlRegex = /(https?:\/\/[^\s]+)/g;
      return text.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
                 .replace(urlRegex, function(url) {
                   return `<a href="${url}" target="_blank" style="color:var(--accent-color); font-weight:700; text-decoration:underline;">${url}</a>`;
                 })
                 .replace(/\n/g, '<br>');
    }

    /* 붙여넣은 HTML 중 외부 불필요 스타일을 지우고 <a> 링크 및 태그 보존 */
    function cleanHTMLForPaste(node) {
      const allowedTags = ['A', 'B', 'STRONG', 'I', 'EM', 'U', 'BR', 'P', 'SPAN', 'DIV'];
      const children = Array.from(node.childNodes);
      children.forEach(child => {
        if (child.nodeType === Node.ELEMENT_NODE) {
          if (!allowedTags.includes(child.tagName)) {
            let text = document.createTextNode(child.textContent);
            child.parentNode.replaceChild(text, child);
          } else {
            child.removeAttribute('style');
            child.removeAttribute('class');
            if (child.tagName === 'A') {
              child.setAttribute('target', '_blank');
              child.style.color = 'var(--accent-color)';
              child.style.fontWeight = '700';
              child.style.textDecoration = 'underline';
            }
            cleanHTMLForPaste(child);
          }
        }
      });
    }

    /* 🎥 스마트 대표 영상 플레이어 지원 */
    function getYouTubeEmbedUrl(url) {
      if (!url) return '';
      let videoId = '';
      if (url.includes('youtu.be/')) {
        videoId = url.split('youtu.be/')[1].split('?')[0].split('&')[0];
      } else if (url.includes('youtube.com/watch')) {
        const urlParams = new URLSearchParams(url.split('?')[1] || '');
        videoId = urlParams.get('v');
      } else if (url.includes('youtube.com/embed/')) {
        return url;
      }
      return videoId ? `https://www.youtube.com/embed/${videoId}` : url;
    }

    function updateVideoPlayer() {
      const inputEl = document.getElementById('video-url-input');
      const typeSelect = document.getElementById('video-type-select');
      if (!inputEl) return;

      const url = inputEl.value.trim();
      const selectedType = typeSelect ? typeSelect.value : 'auto';

      const videoEl = document.getElementById('main-video-element');
      const iframeEl = document.getElementById('main-video-iframe');

      if (!videoEl || !iframeEl) return;

      let isDirectVideo = false;
      let embedUrl = url;

      if (selectedType === 'mp4' || url.match(/\.(mp4|webm|ogg|mov|m3u8)(\?.*)?$/i)) {
        isDirectVideo = true;
      } else if (selectedType === 'iframe') {
        isDirectVideo = false;
        embedUrl = url;
      } else {
        if (url.includes('youtube.com') || url.includes('youtu.be')) {
          embedUrl = getYouTubeEmbedUrl(url);
          isDirectVideo = false;
        } else if (url.match(/\.(mp4|webm|ogg|mov)(\?.*)?$/i)) {
          isDirectVideo = true;
        } else {
          isDirectVideo = false;
          embedUrl = url;
        }
      }

      if (isDirectVideo) {
        iframeEl.style.display = 'none';
        iframeEl.src = '';
        videoEl.style.display = 'block';
        videoEl.src = url;
      } else {
        videoEl.style.display = 'none';
        videoEl.pause();
        iframeEl.style.display = 'block';
        iframeEl.src = embedUrl;
      }

      inputEl.setAttribute('value', url);
    }

    function updateStreamDays() {
      const datePicker = document.getElementById('first-stream-picker');
      const displayBadge = document.getElementById('stream-days-display');
      const textDisplay = document.getElementById('first-stream-text');

      if (!datePicker || !displayBadge) return;

      const startDateStr = datePicker.value;
      if (!startDateStr) return;

      const parts = startDateStr.split('-');
      if (parts.length === 3 && textDisplay) {
        textDisplay.textContent = `${parts[0]}.${parts[1]}.${parts[2]}`;
      }

      const today = new Date();
      today.setHours(0, 0, 0, 0);

      const start = new Date(startDateStr);
      start.setHours(0, 0, 0, 0);

      const diffTime = today - start;
      const diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24)) + 1;

      if (diffDays > 0) {
        displayBadge.textContent = `${diffDays}일차 🎉`;
      } else if (diffDays === 0) {
        displayBadge.textContent = `1일차 🎉`;
      } else {
        displayBadge.textContent = `D${diffDays} (D-${Math.abs(diffDays - 1)})`;
      }
    }

    function syncAllInputsToHTML() {
      document.querySelectorAll('input').forEach(input => {
        if (input.type !== 'file') {
          input.setAttribute('value', input.value);
        }
      });

      document.querySelectorAll('select').forEach(select => {
        Array.from(select.options).forEach(option => {
          if (option.value === select.value) {
            option.setAttribute('selected', 'selected');
            option.selected = true;
          } else {
            option.removeAttribute('selected');
            option.selected = false;
          }
        });
      });

      document.querySelectorAll('textarea').forEach(textarea => {
        textarea.textContent = textarea.value;
      });
    }

      const file = e.target.files[0];
      if (file) {
        if (file.size > 3.5 * 1024 * 1024) {
          alert('⚠️ 움짤/이미지 파일 용량이 너무 큽니다. 3.5MB 이하의 파일을 선택해 주세요.');
          return;
        }
        const reader = new FileReader();
        reader.onload = function(evt) {
          const base64Data = evt.target.result;
          document.getElementById('profile-img-element').src = base64Data;
          try {
            localStorage.setItem('profile_img_src', base64Data);
          } catch(err) {
            alert('⚠️ 용량이 커서 저장에 실패했습니다. 파일 용량이 더 작은 움짤을 사용해 주세요.');
          }
        };
        reader.readAsDataURL(file);
      }
    }

    function calculateDDay(dateStr) {
      if (!dateStr) return 'D-Day';
      const today = new Date(); today.setHours(0,0,0,0);
      const target = new Date(dateStr); target.setHours(0,0,0,0);
      const diffTime = target - today;
      const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
      if (diffDays === 0) return 'D-Day';
      if (diffDays > 0) return `D-${diffDays}`;
      return `D+${Math.abs(diffDays)}`;
    }

    function updateAnnivDate(inputEl) {

      parent.querySelector('.dday-badge').textContent = calculateDDay(selectedDate);
    }

    function renderCalendar() {
      document.getElementById('calendar-title').textContent = `${currentYear}년 ${currentMonth + 1}월`;
      const calBody = document.getElementById('calendar-body'); calBody.innerHTML = '';
      const firstDay = new Date(currentYear, currentMonth, 1).getDay();
      const lastDate = new Date(currentYear, currentMonth + 1, 0).getDate();

      for (let i = 0; i < firstDay; i++) {
        const emptyCell = document.createElement('div'); emptyCell.className = 'cal-cell';
        emptyCell.style.background = 'transparent'; emptyCell.style.border = 'none';
        calBody.appendChild(emptyCell);
      }

      const isAdmin = localStorage.getItem('is_admin_active') === 'true';

      for (let date = 1; date <= lastDate; date++) {
        const cell = document.createElement('div'); cell.className = 'cal-cell';
        const dateKey = `${currentYear}-${currentMonth + 1}-${date}`;
        const cellHeader = document.createElement('div'); cellHeader.className = 'cal-cell-header';
        const num = document.createElement('span'); num.className = 'cal-num'; num.textContent = date;
        cellHeader.appendChild(num);

        if (isAdmin) {
          const addEvtBtn = document.createElement('button'); addEvtBtn.className = 'cal-add-evt-btn'; addEvtBtn.textContent = '+';
          addEvtBtn.onclick = function() { addCalendarEventItem(dateKey, '', ''); };
          cellHeader.appendChild(addEvtBtn);
        }
        cell.appendChild(cellHeader);

        const evtList = document.createElement('div'); evtList.className = 'cal-event-list'; evtList.id = `cal-evt-list-${dateKey}`;
        const events = calendarEvents[dateKey] || [];
        events.forEach((evtObj) => { renderEventItem(evtList, dateKey, evtObj.time, evtObj.text, isAdmin); });

        cell.appendChild(evtList); calBody.appendChild(cell);
      }
    }

    function renderEventItem(container, dateKey, time, text, isAdmin) {
      const item = document.createElement('div'); item.className = 'cal-event-item';
      const topRow = document.createElement('div'); topRow.className = 'cal-event-top';
      const timeSpan = document.createElement('span'); timeSpan.className = 'cal-event-time'; timeSpan.textContent = time;
      timeSpan.contentEditable = isAdmin ? "true" : "false"; timeSpan.oninput = function() { saveCalendarEventsFromDOM(); };
      topRow.appendChild(timeSpan);

      if (isAdmin) {
        const delBtn = document.createElement('button'); delBtn.className = 'cal-event-del'; delBtn.innerHTML = '×';
        delBtn.onclick = function() { item.remove(); saveCalendarEventsFromDOM(); };
        topRow.appendChild(delBtn);
      }
      item.appendChild(topRow);

      const textSpan = document.createElement('span'); textSpan.className = 'cal-event-text'; textSpan.textContent = text || '새 일정';
      textSpan.contentEditable = isAdmin ? "true" : "false"; textSpan.oninput = function() { saveCalendarEventsFromDOM(); };
      item.appendChild(textSpan); container.appendChild(item);
    }

    function addCalendarEventItem(dateKey, time, text) {
      if (!calendarEvents[dateKey]) calendarEvents[dateKey] = [];
      calendarEvents[dateKey].push({ time: time || '', text: text || '새 일정' });
      const container = document.getElementById(`cal-evt-list-${dateKey}`);
      if (container) renderEventItem(container, dateKey, time || '', text || '새 일정', true);
    }

    function saveCalendarEventsFromDOM() {
      const cells = document.querySelectorAll('.cal-cell');
      if (cells.length === 0) return;

      const newCalEvents = {};
      cells.forEach(cell => {
        const listContainer = cell.querySelector('.cal-event-list');
        if (listContainer) {
          const dateKey = listContainer.id.replace('cal-evt-list-', '');
          const items = listContainer.querySelectorAll('.cal-event-item'); const arr = [];
          items.forEach(item => {
            const timeVal = item.querySelector('.cal-event-time').textContent.trim();
            const textVal = item.querySelector('.cal-event-text').textContent.trim();
            if (timeVal || textVal) arr.push({ time: timeVal, text: textVal });
          });
          if (arr.length > 0) newCalEvents[dateKey] = arr;
        }
      });
      calendarEvents = newCalEvents;
    }

    function moveMonth(delta) {
      saveCalendarEventsFromDOM();
      currentMonth += delta;
      if (currentMonth < 0) { currentMonth = 11; currentYear--; }
      else if (currentMonth > 11) { currentMonth = 0; currentYear++; }
      renderCalendar();
    }

        const gamesTab = document.getElementById('tab-games');
        const tmiTab = document.getElementById('tab-tmi');

        if (mainHtmlToSave) localStorage.setItem('sheet_main_html', mainHtmlToSave);
        if (achTab) localStorage.setItem('sheet_achievements_html', achTab.innerHTML);
        if (gamesTab) localStorage.setItem('sheet_games_html', gamesTab.innerHTML);

      } catch (err) {

        if (err.name === 'QuotaExceededError' || err.code === 22) {
          try {
            localStorage.removeItem('sheet_main_html');

    function resetPage() {
      const activeTab = document.querySelector('.tab-content.active');
      if (!activeTab) return;

      let sheetName = '현재';
      let storageKey = '';

      switch(activeTab.id) {
        case 'tab-main':
          sheetName = '🎀 메인 프로필';
          storageKey = 'sheet_main_html';
          break;
        case 'tab-monthly':
          sheetName = '📅 월간 일정 달력';
          storageKey = 'cal_events_time_v2';
          break;
        case 'tab-achievements':
          sheetName = '🏆 방송업적일대기';
          storageKey = 'sheet_achievements_html';
          break;
        case 'tab-games':
          sheetName = '🎮 플레이한 게임';
          storageKey = 'sheet_games_html';
          break;
        case 'tab-tmi':
          sheetName = '💬 TMI & 여담';
          storageKey = 'sheet_tmi_html';
          break;
      }

      if (confirm(`현재 [${sheetName}] 시트의 데이터만 초기화됩니다. 정말 초기화하시겠습니까?`)) {
        if (storageKey) {
          localStorage.removeItem(storageKey);
          if (activeTab.id === 'tab-main') {
            localStorage.removeItem('profile_img_src');
            localStorage.removeItem('saved_video_url');
            localStorage.removeItem('saved_video_type');
          }
        }
        location.reload();
      }
    }
